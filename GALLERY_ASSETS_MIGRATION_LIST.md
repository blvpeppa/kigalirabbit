async function uploadGalleryItem(item) {
  try {
    const filePath = path.join(ASSETS_DIR, item.file);
    
    if (!fs.existsSync(filePath)) {
      console.error(`File not found: ${filePath}`);
      return null;
    }

    const formData = new FormData();
    formData.append('file', fs.createReadStream(filePath));
    formData.append('title', item.title);
    formData.append('alt_text', item.alt_text);
    formData.append('type', item.type);
    formData.append('display_order', item.order.toString());

    const response = await axios.post(API_BASE, formData, {
      headers: {
        ...formData.getHeaders(),
      },
    });

    if (response.data.success) {
      console.log(`✅ Uploaded: ${item.file} (ID: ${response.data.data.id})`);
      return response.data.data;
    } else {
      console.error(`❌ Failed: ${item.file} - ${response.data.message}`);
      return null;
    }
  } catch (error) {
    console.error(`❌ Error uploading ${item.file}:`, error.message);
    return null;
  }
}

async function uploadAll() {
  console.log('🚀 Starting gallery migration...\n');
  
  for (const item of galleryItems) {
    await uploadGalleryItem(item);
    // Small delay to avoid overwhelming the server
    await new Promise(resolve => setTimeout(resolve, 500));
  }
  
  console.log('\n✨ Migration complete!');
}

uploadAll();
